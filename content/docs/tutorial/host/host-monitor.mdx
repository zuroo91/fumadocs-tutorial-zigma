### 1. Validasi Keamanan Host & Proteksi Navigasi
Sebelum merender map dan memantau room, sistem memastikan user yang membuka halaman ini adalah benar-benar host yang sah:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
private _setupScene() {
    SessionStore.set('gameStarted', 'true', 'host');
    this.registry.set('room', this.room);
    const profile = authService.getStoredProfile();
    const hostUserId = (this.room as any)?.metadata?.hostId;
    if (profile && hostUserId) {
        const identityCandidates = [profile.id, profile.auth_user_id]
            .filter((v): v is string => typeof v === 'string' && v.trim().length > 0)
            .map(v => String(v).trim().toLowerCase());
        const hostIdNorm = String(hostUserId).trim().toLowerCase();
        
        // Blokir jika user bukan host dari room ini
        if (!identityCandidates.includes(hostIdNorm)) {
            console.error("[Spectator] Security block: Current logged-in user is not the host of this room!");
            window.location.replace('/');
            return;
        }
    }
    
    // Cegah tombol back browser merusak game
    const currentUrl = window.location.pathname;
    window.history.pushState({ zigmaHostProgress: true }, '', currentUrl);
    this._popStateHandler = () => {
        window.history.pushState({}, '', currentUrl);
        this.showEndGamePopup(); // Tampilkan popup konfirmasi akhiri kuis jika menekan back
    };
    window.addEventListener('popstate', this._popStateHandler);
}
```

### 2. Sinkronisasi State Colyseus & Auto-Recovery
Host memantau event dari Colyseus room seperti waktu countdown awal, sinkronisasi timer kuis, dan mekanisme auto-reconnect tanpa kehilangan data:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
private _registerRoomListeners(room: Room) {
    // 1. Timer update dari server
    room.onMessage('timerUpdate', (data: { remaining: number }) => {
        this.updateTimer(data.remaining);
    });
    // 2. Countdown 3-2-1 Go
    room.state.listen("countdown", (val: number) => {
        if (val > 0) {
            TransitionManager.setCountdownText(val.toString());
        } else if (val === 0) {
            TransitionManager.setCountdownText(i18n.t('game.go'));
        }
    });
    // 3. Robust Reconnection jika koneksi host terputus sementara
    room.onLeave((code) => {
        if (this.isEnding) return;
        if (code !== 1000 && !this.isTransitioning) {
            restoreRoomConnection({
                client,
                role: 'host',
                isHost: true,
                roomId: savedRoomId,
                onStatus: (msg) => showRestoreBanner(msg),
            }).then((newRoom) => {
                if (newRoom) this.scene.restart({ room: newRoom });
            });
        }
    });
}
```

### 3. Kontrol Kamera Host (Free-Pan & Zoom)
Host dapat memperbesar, memperkecil, atau menggeser (*drag/pan*) kamera untuk mengawasi sudut peta tertentu:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
// Zoom menggunakan scroll wheel mouse
this.input.on('wheel', (pointer, gameObjects, deltaX, deltaY) => {
    const newZoom = this.cameras.main.zoom - (deltaY * 0.001);
    this.cameras.main.setZoom(Phaser.Math.Clamp(newZoom, this.minZoom, 3.0));
});
// Drag & Pan kamera dengan klik-tahan mouse
this.input.on('pointerdown', (pointer) => {
    if (pointer.button === 2) return;
    this.isDragPan = true;
    this.dragOrigin = new Phaser.Math.Vector2(pointer.x, pointer.y);
});
this.input.on('pointermove', (pointer) => {
    if (this.isDragPan && this.dragOrigin) {
        const deltaX = pointer.x - this.dragOrigin.x;
        const deltaY = pointer.y - this.dragOrigin.y;
        this.cameras.main.scrollX -= deltaX / this.cameras.main.zoom;
        this.cameras.main.scrollY -= deltaY / this.cameras.main.zoom;
        this.dragOrigin.set(pointer.x, pointer.y);
    }
});
this.input.on('pointerup', () => this.isDragPan = false);
```

### 4. Pelacakan Pemain & Progress Bar Real-time
Setiap pemain yang masuk (`players.onAdd`) dirender sebagai container berisi sprite karakter, rambut, nama, serta progress bar yang menghitung rasio soal yang sudah dijawab:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
const handlePlayerAdd = (player: any, sessionId: string) => {
    if (player.isHost) return; // Host tidak memiliki avatar playable
    const container = this.add.container(player.x, player.y).setScale(2.5);
    const baseSprite = this.add.sprite(0, 0, 'character').play('idle');
    const toolsSprite = this.add.sprite(0, 0, 'tools_idle').play('tools_idle');
    const hairSprite = this.add.sprite(0, 0, '').setVisible(false);
    container.add([baseSprite, toolsSprite, hairSprite]);
    this.createNameTag(sessionId, player.name || 'Player', container);
    this.playerEntities[sessionId] = container;

    // Kalkulasi dan render progres kuis pemain
    const updateProgress = () => {
        const progressBar = container.getData('progressBar') as Phaser.GameObjects.Graphics;
        const answered = player.answeredQuestions || 0;
        const qLimit = this.room.state.questionLimit;
        const target = (qLimit === 'all' || !qLimit) 
            ? this.room.state.totalQuestions 
            : parseInt(qLimit) || 5;
        const progress = Phaser.Math.Clamp(answered / target, 0, 1);
        if (progressBar) {
            progressBar.clear();
            // Background bar
            progressBar.fillStyle(0x000000, 0.5);
            progressBar.fillRect(0, 0, 36, 5);
            // Hijau terang menunjukkan rasio progres
            progressBar.fillStyle(0x00ff88, 1);
            progressBar.fillRect(0, 0, 36 * progress, 5);
            progressBar.lineStyle(1, 0x000000, 1);
            progressBar.strokeRect(0, 0, 36, 5);
        }
    };

    // Dengarkan perubahan state tiap ada soal yang dijawab / pergerakan
    this.disposers.push(player.onChange(() => {
        container.setData({ targetX: player.x, targetY: player.y });
        updateHair();
        updateProgress();
    }));
};
```

### 5. Render Loop & Interpolasi Animasi (`update`)
Di setiap frame (`update`), posisi karakter dihaluskan menggunakan interpolasi LERP (`container.x += (tx - container.x) * 0.25`) dan status animasi diganti secara dinamis antara `attack`, `walk`, atau `idle`:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
update(time: number, delta: number) {
    if (!this.room || !this.room.state) return;
    Object.keys(this.playerEntities).forEach(sessionId => {
        const container = this.playerEntities[sessionId];
        const player = this.room.state.players.get(sessionId);
        const tx = container.getData('targetX');
        const ty = container.getData('targetY');
        if (tx !== undefined && ty !== undefined) {
            // Interpolasi posisi mulus
            container.x += (tx - container.x) * 0.25;
            container.y += (ty - container.y) * 0.25;
            const dx = tx - container.x;
            const dy = ty - container.y;
            const isMoving = Math.abs(dx) > 0.5 || Math.abs(dy) > 0.5;

            // State animasi: Attack saat menebas pohon/batu soal kuis
            if (player?.isAttacking) {
                base.play('base_attack', true);
                tools.play('tools_attack', true);
            } else if (isMoving) {
                base.play('walk', true);
                tools.play('tools_walk', true);
                // Flip arah pandang horizontal
                if (Math.abs(dx) > 0.1) {
                    base.setFlipX(dx < 0);
                    tools.setFlipX(dx < 0);
                }
            } else {
                base.play('idle', true);
                tools.play('tools_idle', true);
            }
        }
    });
}
```

### 6. Kontrol Akhiri Permainan (End Game by Host)
Host memiliki tombol darurat di pojok kanan bawah untuk mengakhiri kuis sebelum timer habis. Saat dikonfirmasi, pesan dikirim ke Colyseus server:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
showEndGamePopup() {
    // ...menampilkan modal konfirmasi "Akhiri Permainan?"...
    yesBtn.onclick = () => {
        yesBtn.disabled = true;
        // Kirim event ke Colyseus room
        if (this.room && this.room.connection.isOpen) {
            this.room.send('hostEndGame');
        }
        closePopup();
    };
}
```

Ketika server merespon dengan pesan `gameEnded`, host otomatis diarahkan ke halaman Leaderboard:

```typescript title="client/src/scenes/host/[roomCode]/progress/page.ts"
room.onMessage('gameEnded', (data: any) => {
```
