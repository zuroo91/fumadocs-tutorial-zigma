# Membangun Leaderboard & Podium Ranking

Pada modul ini, kamu akan membangun tampilan klasemen akhir (`HostLeaderboardManager`) yang menampilkan animasi selebrasi podium tiga juara teratas, rekapitulasi nilai seluruh peserta, serta penyimpanan permanen ke database Supabase.

---

## Langkah 1: Membangun Komponen Tampilan Leaderboard Host

Buat file `client/src/scenes/host/HostLeaderboardManager.ts`:

```typescript title="client/src/scenes/host/HostLeaderboardManager.ts"
import { RoomService } from '../../services/room/RoomService';

export interface LeaderboardEntry {
  rank: number;
  nickname: string;
  score: number;
  answeredCount: number;
}

export class HostLeaderboardManager {
  private container: HTMLDivElement;
  private roomService = RoomService.getInstance();

  constructor() {
    this.container = document.createElement('div');
    this.container.id = 'host-leaderboard-view';
    this.container.className = 'fixed inset-0 bg-gray-950 text-white flex flex-col p-8 z-30 overflow-y-auto';

    this.render();
  }

  private render() {
    const room = this.roomService.getActiveRoom();
    if (!room) return;

    // Kumpulkan data pemain dan urutkan berdasarkan skor tertinggi
    const entries: LeaderboardEntry[] = [];
    room.state.players.forEach((player) => {
      if (!player.isHost) {
        entries.push({
          rank: 1,
          nickname: player.nickname,
          score: player.score,
          answeredCount: player.answeredCount
        });
      }
    });

    entries.sort((a, b) => b.score - a.score);
    entries.forEach((e, idx) => e.rank = idx + 1);

    const top1 = entries[0] || null;
    const top2 = entries[1] || null;
    const top3 = entries[2] || null;

    this.container.innerHTML = `
      <div class="max-w-4xl mx-auto w-full text-center">
        <h1 class="text-4xl font-extrabold text-transparent bg-clip-text bg-gradient-to-r from-yellow-400 via-amber-300 to-yellow-500 mb-8">
          HASIL AKHIR PERTANDINGAN 🏆
        </h1>

        <!-- Podium 3 Besar -->
        <div class="grid grid-cols-3 gap-4 items-end mb-12 max-w-lg mx-auto">
          <!-- Juara 2 -->
          <div class="bg-gray-900 border border-gray-700 p-4 rounded-2xl h-40 flex flex-col justify-end">
            <span class="text-2xl font-black text-gray-300">🥈 2nd</span>
            <span class="font-bold truncate mt-2">${top2 ? top2.nickname : '-'}</span>
            <span class="text-sm text-teal-400 font-mono">${top2 ? `${top2.score} Pts` : ''}</span>
          </div>
          <!-- Juara 1 -->
          <div class="bg-gradient-to-b from-yellow-900/60 to-gray-900 border border-yellow-500/50 p-4 rounded-2xl h-52 flex flex-col justify-end shadow-xl">
            <span class="text-4xl font-black text-yellow-400">🥇 1st</span>
            <span class="font-extrabold text-lg truncate mt-2">${top1 ? top1.nickname : '-'}</span>
            <span class="text-base text-yellow-300 font-mono font-bold">${top1 ? `${top1.score} Pts` : ''}</span>
          </div>
          <!-- Juara 3 -->
          <div class="bg-gray-900 border border-gray-700 p-4 rounded-2xl h-32 flex flex-col justify-end">
            <span class="text-2xl font-black text-amber-600">🥉 3rd</span>
            <span class="font-bold truncate mt-2">${top3 ? top3.nickname : '-'}</span>
            <span class="text-sm text-teal-400 font-mono">${top3 ? `${top3.score} Pts` : ''}</span>
          </div>
        </div>

        <!-- Tabel Klasemen Lengkap -->
        <div class="bg-gray-900 border border-gray-800 rounded-2xl overflow-hidden mb-8">
          <table class="w-full text-left text-sm">
            <thead class="bg-gray-800 text-gray-400 text-xs uppercase">
              <tr>
                <th class="p-3">Rank</th>
                <th class="p-3">Nickname</th>
                <th class="p-3 text-center">Soal Terjawab</th>
                <th class="p-3 text-right">Total Skor</th>
              </tr>
            </thead>
            <tbody class="divide-y divide-gray-800">
              ${entries.map(e => `
                <tr class="hover:bg-gray-800/50">
                  <td class="p-3 font-bold text-gray-400">#${e.rank}</td>
                  <td class="p-3 font-semibold">${e.nickname}</td>
                  <td class="p-3 text-center">${e.answeredCount}</td>
                  <td class="p-3 text-right font-mono font-bold text-teal-300">${e.score}</td>
                </tr>
              `).join('')}
            </tbody>
          </table>
        </div>

        <!-- Tombol Aksi -->
        <button id="btn-back-home" class="px-8 py-3 bg-purple-600 hover:bg-purple-500 rounded-xl font-bold transition">
          Kembali ke Beranda
        </button>
      </div>
    `;

    document.body.appendChild(this.container);

    document.getElementById('btn-back-home')?.addEventListener('click', () => {
      this.destroy();
      window.location.hash = '#/';
    });
  }

  public destroy() {
    this.container.remove();
  }
}
```

* **File dibuat**: `client/src/scenes/host/HostLeaderboardManager.ts`
* **Fungsi**: Merender podium interaktif dan tabel hasil akhir kuis.
